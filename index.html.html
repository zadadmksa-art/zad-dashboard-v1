/** * زاد التنموية — Web App API * يوفر رابط JSON + JSONP لربط الداشبورد بـ Google Sheets * ✅ يدعم JSONP لحل مشكلة CORS */ function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const output = { associations: [], achievements: [], targeted: [], alerts: [] }; // ═══════════════════════════════════════ // 1. ورقة: الجمعيات_المتعاقدة // ═══════════════════════════════════════ try { const sheet = ss.getSheetByName("الجمعيات_المتعاقدة"); const data = sheet.getRange(3, 1, sheet.getLastRow() - 2, 9).getValues(); data.forEach((row, i) => { const name = String(row[0]).trim(); if (!name || name === "الإجمالي") return; output.associations.push({ id: i, name: name, contractStart: formatDate(row[1]), contractEnd: formatDate(row[2]), contract: toNum(row[3]), paid: toNum(row[4]), beneficiaries: toNum(row[6]), grants: toNum(row[7]), programs: toNum(row[8]) }); }); } catch(err) { output.error_associations = err.message; } // ═══════════════════════════════════════ // 2. ورقة: الإنجازات // ═══════════════════════════════════════ try { const sheet = ss.getSheetByName("الإنجازات"); const lastRow = sheet.getLastRow(); if (lastRow >= 4) { const data = sheet.getRange(4, 1, lastRow - 3, 4).getValues(); data.forEach(row => { const assoc = String(row[0]).trim(); if (!assoc) return; output.achievements.push({ assoc: assoc, service: String(row[1]).trim(), desc: String(row[2]).trim(), date: formatDate(row[3]) }); }); } } catch(err) { output.error_achievements = err.message; } // ═══════════════════════════════════════ // 3. ورقة: الجمعيات_المستهدفة // ═══════════════════════════════════════ try { const sheet = ss.getSheetByName("الجمعيات_المستهدفة"); const lastRow = sheet.getLastRow(); if (lastRow >= 3) { const data = sheet.getRange(3, 1, lastRow - 2, 2).getValues(); data.forEach(row => { const name = String(row[0]).trim(); if (!name) return; output.targeted.push({ name: name, status: String(row[1]).trim() }); }); } } catch(err) { output.error_targeted = err.message; } // ═══════════════════════════════════════ // 4. ورقة: التنبيهات // ═══════════════════════════════════════ try { const sheet = ss.getSheetByName("التنبيهات"); const lastRow = sheet.getLastRow(); if (lastRow >= 4) { const data = sheet.getRange(4, 1, lastRow - 3, 3).getValues(); data.forEach(row => { const title = String(row[0]).trim(); if (!title) return; output.alerts.push({ title: title, desc: String(row[1]).trim(), date: formatDate(row[2]) }); }); } } catch(err) { output.error_alerts = err.message; } // ═══════════════════════════════════════ // ✅ إرجاع JSON أو JSONP حسب وجود callback // ═══════════════════════════════════════ const jsonString = JSON.stringify(output); const callback = e && e.parameter && e.parameter.callback; if (callback) { // JSONP — يحل مشكلة CORS في المتصفح return ContentService .createTextOutput(callback + '(' + jsonString + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } // JSON عادي return ContentService .createTextOutput(jsonString) .setMimeType(ContentService.MimeType.JSON); } /* ───────────────────────────────── مساعدات ───────────────────────────────── */ function toNum(val) { if (val === "" || val === null || val === undefined) return 0; const n = parseFloat(String(val).replace(/,/g, "")); return isNaN(n) ? 0 : n; } function formatDate(val) { if (!val) return ""; if (val instanceof Date) { const y = val.getFullYear(); const m = String(val.getMonth() + 1).padStart(2, "0"); const d = String(val.getDate()).padStart(2, "0"); return `${y}/${m}/${d}`; } return String(val).trim(); }
